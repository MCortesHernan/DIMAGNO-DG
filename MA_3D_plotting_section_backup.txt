from matplotlib.ticker import MaxNLocator, StrMethodFormatter, LogFormatterMathtext

import nmmn.plots
import matplotlib.colors as colors
parula = nmmn.plots.parulacmap()


class plot_class:
    '''This class defines methods to make different kinds of plots.'''

    def __init__(self, dim, title,
                 z_title=r'$z$ [cm]', x_title=r'$r$ [cm]',
                 fig=None, ax=None):

        plt.rcParams['font.family'] = 'sans-serif'
        self.dim = dim
        self.title = title
        self.z_title = z_title
        self.x_title = x_title

        # Use existing fig/ax or create new
        if fig is None or ax is None:
            self.fig, self.ax = plt.subplots()
        else:
            self.fig = fig
            self.ax = ax
        if dim == 2:
            self.ax.set_box_aspect(None)

        self.ax.tick_params(labelsize=30)

        self.ax.xaxis.set_major_formatter(StrMethodFormatter('{x:g}'))
        self.ax.yaxis.set_major_formatter(StrMethodFormatter('{x:g}'))

    def plot_map(self, X, Z, data, clip=None, log=False, add_colorbar=False, cax=None, write_label_x=False, write_label_y=False):
        '''2D colormap. Optionally attach colorbar.'''

        data = np.array(data, copy=True)

        # Sanitize
        data[~np.isfinite(data)] = np.nan

        if log:
            data[data <= 0] = np.nan
            if clip is None:
                normalize = colors.LogNorm(vmin=np.nanmin(data), vmax=np.nanmax(data))
            else:
                normalize = colors.LogNorm(clip[0], clip[1])
        else:
            normalize = None if clip is None else colors.Normalize(clip[0], clip[1])

        pcol = self.ax.pcolormesh(X, Z, data, cmap=parula,
                                  norm=normalize, shading='auto')

        if add_colorbar:
            cbar = self.fig.colorbar(pcol, cax=cax)
            cbar.ax.tick_params(labelsize=30)

            if self.title is not None:
                self.ax.set_title(self.title, fontsize=30, pad=10)

        self.ax.tick_params('x', direction= 'in', length = 5, top = True)
        self.ax.tick_params('y', direction= 'in', length = 5, right = True)
        self.ax.set_xlim(np.nanmin(X), np.nanmax(X))
        self.ax.set_ylim(np.nanmin(Z), np.nanmax(Z))

        if write_label_x:
            self.ax.set_xlabel(self.z_title, fontsize=30)
        if write_label_y:
            self.ax.set_ylabel(self.x_title, fontsize=30)

        return pcol

    def plot_line(self, x_s, magnitude, label, color=['black'], linestyle=['solid'], linewidth=[1.], logy=False, write_label_x=False, write_label_y=False, add_legend=True, lim_x=None, lim_y=None):
        if not isinstance(magnitude, list):
            magnitude = [magnitude]
            label = [label]

        for i in range(len(magnitude)):
            self.ax.plot(x_s, magnitude[i], label=label[i], color=color[i], linestyle=linestyle[i], linewidth=linewidth[i])

        if logy:
            # Avoid zeros or negatives
            all_values = np.hstack(magnitude)
            min_val = max(np.min(all_values[all_values > 0]), 1e-12)  # avoid log(0)
            max_val = np.max(all_values)

            self.ax.set_yscale('log')
            self.ax.set_ylim(min_val, max_val)

            # Format y-axis labels as powers of 10
            self.ax.yaxis.set_major_formatter(LogFormatterMathtext(base=10))

        self.ax.tick_params('x', direction= 'in', length = 5, top = True)
        self.ax.tick_params('y', direction= 'in', length = 5, right = True)

        if write_label_x:
            self.ax.set_xlabel(self.z_title, fontsize=30)
        if write_label_y:
            self.ax.set_ylabel(self.x_title, fontsize=30)

        if add_legend:
            self.ax.legend(fontsize=30)

        #set lims
        if lim_y is not None:
            self.ax.set_ylim(lim_y)
        else:
            self.ax.set_ylim([np.min(magnitude), np.max(magnitude) + 0.06*np.max(magnitude)])
        if lim_x is not None:
            self.ax.set_xlim(lim_x)
        else:
            self.ax.set_xlim([np.min(x_s), np.max(x_s)])

def vac(S, data):
    '''data where n > n_plot_min, NaN elsewhere (blank in the plots).'''
    return np.where(S['n'] > n_plot_min, data, np.nan)

def je_streams(S, comps, frac=1e-3):
    '''In-plane electron current j_e = -n u_e (comps, e.g. ('uez', 'uex')), masked where |j_e| < frac max.'''
    U, V = -S['n_e']*S[comps[0]], -S['n_e']*S[comps[1]]
    m = np.hypot(U, V) < frac*np.nanmax(np.hypot(U, V))
    return np.ma.masked_where(m, U), np.ma.masked_where(m, V)

def log_clip(data, pct=1):
    '''Log color limits: [pct percentile, max] of the positive finite values.'''
    d = np.abs(np.asarray(data)); d = d[np.isfinite(d) & (d > 0)]
    return [np.percentile(d, pct), d.max()]

def lin_clip(data, lo=1, hi=99):
    '''Linear color limits: [lo, hi] percentiles of the finite values.'''
    d = np.asarray(data); d = d[np.isfinite(d)]
    return [np.percentile(d, lo), np.percentile(d, hi)] if d.size else [0., 1.]

def reflect_about_y(field, y_lin, y_line):
    '''Reflects field(z, y) about the line y = y_line (interpolation).'''
    y_reflected_coords = 2*y_line - y_lin
    return np.array([np.interp(y_reflected_coords, y_lin, row, left=np.nan, right=np.nan) for row in field])

def asymmetry_about_y(field, y_lin, y_line, parity=+1):
    '''
    parity=+1: field(y) = field(2*y_line - y) expected (scalars, z component)
    parity=-1: field(y) = -field(2*y_line - y) expected (y component)
    Returns the asymmetry residual (0 for a perfectly symmetric field).
    '''
    reflected = reflect_about_y(field, y_lin, y_line)
    return 0.5 * (field - parity * reflected)

def add_contour(plot, key):
    '''3D magnetic lines (black) and separatrix (red) on the plane key: 'y0', 'x0' or a z slice value.'''
    xlim, ylim = plot.ax.get_xlim(), plot.ax.get_ylim()
    if key == 'y0':
        for L in lines_y0:
            plot.ax.plot(L[:, 0], L[:, 1], color='black', linestyle='solid', linewidth=1.5)
        plot.ax.plot(sep_y0[:, 0], sep_y0[:, 1], color='red', linestyle='solid', linewidth=3.)
    elif key == 'x0':
        for L in lines_x0:   # projection of the 3D lines on the plane
            plot.ax.plot(L[:, 0], L[:, 2], color='black', linestyle='solid', linewidth=1.5)
        plot.ax.contour(Zt_x0, Yt_x0, topo_x0, [0.], colors='red', linestyles='solid', linewidths=3.)
    else:                    # z slice: horizontal axis y, vertical axis x
        plot.ax.contour(Yt_z, Xt_z, topo_z[key], [0.], colors='red', linestyles='solid', linewidths=3.)
    plot.ax.set_xlim(xlim)
    plot.ax.set_ylim(ylim)

def add_streams(ax, Hg, Vg, U, V):
    '''Magenta streamlines of (U, V) drawn on the grid (Hg, Vg).'''
    if np.ptp(Hg[:, 0]) > 0:  # streamplot wants the horizontal coordinate along axis 1
        Hg, Vg, U, V = Hg.T, Vg.T, U.T, V.T
    ax.streamplot(Hg[0, :], Vg[:, 0], U, V, color='magenta', density=0.7, arrowsize=1., linewidth = 1.)

def tick_number(plot, nx = 4, ny = 4):
    plot.ax.xaxis.set_major_locator(MaxNLocator(nbins=nx))  # max nx ticks on X
    plot.ax.yaxis.set_major_locator(MaxNLocator(nbins=ny))  # max ny ticks on Y

def plot_panels(name, shape, figsize, panels, Hg, Vg, h_title=z_title, v_title=x_title,
                contour=None, stream=None, aspect=None, sharex=False, sharey=True, wspace=0.04):
    '''
    One figure with a grid of 2D maps, saved as plot_mesh_dir/<name>.png
    panels : list of dicts with 'data' and 'title', and optionally 'log', 'clip' or any of the
             keyword arguments below (Hg, Vg, v_title, contour, stream, ...) to override them for that panel.
    Hg, Vg : 2D grids of the horizontal and vertical axis.
    contour: plane of the 3D lines and separatrix ('y0', 'x0', a z slice value) or None.
    stream : (U, V) components drawn as streamlines, or None.
    Axis labels are written on the bottom row / first column only when the axis is shared.
    '''
    fig, axs = plt.subplots(*shape, figsize=figsize, sharex=sharex, sharey=sharey, squeeze=False)
    fig.subplots_adjust(left=0.03, right=0.98, bottom=0.04, top=0.95, wspace=wspace, hspace=0.1)

    defaults = dict(log=False, clip=None, Hg=Hg, Vg=Vg, h_title=h_title, v_title=v_title,
                    contour=contour, stream=stream, aspect=aspect)

    for k, (ax, panel) in enumerate(zip(axs.flat, panels)):
        o = {**defaults, **panel}
        row, col = divmod(k, shape[1])

        p = plot_class(2, o['title'], fig=fig, ax=ax, z_title=o['h_title'], x_title=o['v_title'])
        p.plot_map(o['Hg'], o['Vg'], o['data'], clip=o['clip'], log=o['log'], add_colorbar=True,
                   write_label_x=(not sharex or row == shape[0] - 1),
                   write_label_y=(not sharey or col == 0))
        if o['contour'] is not None:
            add_contour(p, o['contour'])
        if o['stream'] is not None:
            add_streams(p.ax, o['Hg'], o['Vg'], *o['stream'])
        if o['aspect'] is not None:
            p.ax.set_aspect(o['aspect'])

    fig.savefig(os.path.join(plot_mesh_dir, name + '.png'), dpi=500, bbox_inches='tight')
    return fig

def prop_panels(S, stream):
    '''n, phi, Te and u_i panels (streamlines on the u_i panel).'''
    return [dict(data=S['n']*n0_m3,     title=r'$n$ $[m^{-3}]$', log=True),
            dict(data=S['phi']*Te0,     title=r'$\phi$ $[V]$',   clip=[np.nanmin(S['phi']*Te0), 0]),
            dict(data=S['Te']*Te0,      title=r'$T_e$ $[eV]$',   clip=[0, 10]),
            dict(data=S['ui']*u_scale,  title=r'${\tilde{u}}_{i}$ $[km/s]$', stream=stream)]

def xyz_panels(S, keys, titles, clips, scale=1.):
    '''|z|, |x|, |y| components of a vector, log scale.'''
    return [dict(data=np.abs(S[k]*scale), title=t, log=True, clip=c) for k, t, c in zip(keys, titles, clips)]
